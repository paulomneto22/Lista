namespace Lista
{
    internal class Program
    {
        static void Main(string[] args)
        {
            List<double> P1 = new List<double> { 7, 8, 6, 9 };
            List<double> P2 = new List<double> { 5, 7, 8, 6 };
 
            double somaP1 = 0;
            double somaP2 = 0;
 
            for (int i = 0; i < P1.Count; i++)
            {
                somaP1 += P1[i];
                somaP2 += P2[i];
            }
 
            double mediaP1 = somaP1 / P1.Count;
            double mediaP2 = somaP2 / P2.Count;
 
            Console.WriteLine("Média da Prova 1: " + mediaP1);
            Console.WriteLine("Média da Prova 2: " + mediaP2);
 
            if (mediaP1 > mediaP2)
            {
                Console.WriteLine("A turma teve a melhor média na Prova 1.");
            }
            else if (mediaP2 > mediaP1)
            {
                Console.WriteLine("A turma teve a melhor média na Prova 2.");
            }
            else
            {
                Console.WriteLine("As duas provas tiveram a mesma média.");
            }
        }
    }
}
