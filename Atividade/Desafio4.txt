namespace Lista
{
    internal class Program
    {
        static void Main(string[] args)
        {
            List<int> numeros = new List<int>();
            List<int> semRepeticao = new List<int>();
            List<int> repetidos = new List<int>();

            while (true)
            {
                Console.Write($"Informe um número (zero para sair): ");
                string entrada = Console.ReadLine();

                if (entrada == "0")
                    break;

                if (!int.TryParse(entrada, out int numero))
                {
                    Console.WriteLine($"Informe apenas números!");
                    continue;
                }

                numeros.Add(numero);
            }

            foreach (int numero in numeros)
            {
                int quantidade = 0;

                foreach (int n in numeros)
                {
                    if (n == numero)
                        quantidade++;
                }

                if (quantidade == 1)
                    semRepeticao.Add(numero);
                else if (!repetidos.Contains(numero))
                    repetidos.Add(numero);
            }

            Console.WriteLine($"Números informados: [{string.Join(", ", numeros)}]");
            Console.WriteLine($"Números que não repetiram: [{string.Join(", ", semRepeticao)}]");
            Console.WriteLine($"Somente números que se repetiram: [{string.Join(", ", repetidos)}]");
        }
    }
}
